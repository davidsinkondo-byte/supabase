create extension if not exists btree_gist;

-- ---------------------------------------------------------------------
-- Gestionnaires autorisés (liste blanche par e-mail).
-- Un compte Supabase dont l'e-mail n'est pas ici ne voit AUCUNE donnée,
-- même si les inscriptions publiques ont été laissées ouvertes par erreur.
-- ---------------------------------------------------------------------
create table if not exists public.gestionnaires (
  email      text primary key,
  nom        text not null default '',
  created_at timestamptz not null default now()
);
alter table public.gestionnaires enable row level security;  -- aucune policy : invisible depuis l'application
revoke all on public.gestionnaires from anon, authenticated;

create or replace function public.is_gestionnaire() returns boolean
language sql stable security definer set search_path = public as $$
  select exists (
    select 1 from public.gestionnaires g
    where lower(g.email) = lower(coalesce(auth.jwt() ->> 'email', ''))
  );
$$;

create or replace function public.set_updated_at() returns trigger
language plpgsql as $$ begin new.updated_at := now(); return new; end $$;

-- ---------------------------------------------------------------------
-- Agents
-- ---------------------------------------------------------------------
create table if not exists public.agents (
  id         text primary key,                 -- = matricule (ex. « 2274 W »)
  matricule  text not null unique,
  nom        text not null,
  prenom     text not null default '',
  fonction   text not null default '',
  direction  text not null,
  service    text not null default '',
  solde      integer not null default 30,      -- jours de congé annuel restants
  updated_at timestamptz not null default now()
);
drop trigger if exists agents_updated_at on public.agents;
create trigger agents_updated_at before update on public.agents
  for each row execute function public.set_updated_at();

alter table public.agents enable row level security;
revoke all on public.agents from anon;
grant select, insert, update, delete on public.agents to authenticated;
drop policy if exists agents_gestionnaires on public.agents;
create policy agents_gestionnaires on public.agents for all to authenticated
  using (public.is_gestionnaire()) with check (public.is_gestionnaire());

-- ---------------------------------------------------------------------
-- Demandes de congé
-- ---------------------------------------------------------------------
create table if not exists public.demandes (
  id          uuid primary key default gen_random_uuid(),
  agent_id    text not null references public.agents(id) on update cascade on delete restrict,
  type        text not null check (type in ('annuel', 'exceptionnel', 'maternite')),
  du          date not null,
  au          date not null,
  duree       integer not null check (duree > 0),
  motif       text not null default '',
  interim     text not null default '',
  statut      text not null default 'attente' check (statut in ('attente', 'approuve', 'refuse')),
  solde_apres integer,
  motif_refus text not null default '',
  created_at  timestamptz not null default now(),
  created_by  uuid default auth.uid(),
  decided_at  date,
  decided_by  uuid,
  updated_at  timestamptz not null default now(),
  constraint demandes_dates check (au >= du),
  -- Deux demandes non refusées d'un même agent ne peuvent pas se chevaucher,
  -- même si deux gestionnaires les saisissent au même moment.
  constraint demandes_sans_chevauchement exclude using gist (
    agent_id with =, daterange(du, au, '[]') with &&
  ) where (statut <> 'refuse')
);
create index if not exists demandes_agent_idx on public.demandes (agent_id);
drop trigger if exists demandes_updated_at on public.demandes;
create trigger demandes_updated_at before update on public.demandes
  for each row execute function public.set_updated_at();

alter table public.demandes enable row level security;
revoke all on public.demandes from anon, authenticated;
-- Saisie et modification : uniquement les champs de la demande.
-- Statut, solde et décision ne changent que via les fonctions ci-dessous.
grant select on public.demandes to authenticated;
grant insert (agent_id, type, du, au, duree, motif, interim) on public.demandes to authenticated;
grant update (agent_id, type, du, au, duree, motif, interim) on public.demandes to authenticated;

drop policy if exists demandes_lecture on public.demandes;
create policy demandes_lecture on public.demandes for select to authenticated
  using (public.is_gestionnaire());
drop policy if exists demandes_saisie on public.demandes;
create policy demandes_saisie on public.demandes for insert to authenticated
  with check (public.is_gestionnaire());
drop policy if exists demandes_modification on public.demandes;
create policy demandes_modification on public.demandes for update to authenticated
  using (public.is_gestionnaire() and statut = 'attente')
  with check (public.is_gestionnaire() and statut = 'attente');

-- ---------------------------------------------------------------------
-- Décisions (atomiques : statut + solde de l'agent dans la même transaction)
-- ---------------------------------------------------------------------
create or replace function public.approuver_demande(p_id uuid) returns void
language plpgsql security definer set search_path = public as $$
declare d public.demandes; v_solde integer;
begin
  if not public.is_gestionnaire() then raise exception 'Accès refusé.' using errcode = '42501'; end if;
  select * into d from public.demandes where id = p_id for update;
  if not found then raise exception 'Demande introuvable (supprimée entre-temps ?).'; end if;
  if d.statut <> 'attente' then raise exception 'Cette demande a déjà été traitée par un autre gestionnaire.'; end if;
  if d.type = 'annuel' then
    update public.agents set solde = solde - d.duree where id = d.agent_id returning solde into v_solde;
  end if;
  update public.demandes
     set statut = 'approuve', solde_apres = v_solde, decided_at = current_date,
         decided_by = auth.uid(), motif_refus = ''
   where id = p_id;
end $$;

create or replace function public.refuser_demande(p_id uuid, p_motif text default '') returns void
language plpgsql security definer set search_path = public as $$
declare d public.demandes; v_solde integer;
begin
  if not public.is_gestionnaire() then raise exception 'Accès refusé.' using errcode = '42501'; end if;
  select * into d from public.demandes where id = p_id for update;
  if not found then raise exception 'Demande introuvable (supprimée entre-temps ?).'; end if;
  if d.statut <> 'attente' then raise exception 'Cette demande a déjà été traitée par un autre gestionnaire.'; end if;
  if d.type = 'annuel' then
    select solde - d.duree into v_solde from public.agents where id = d.agent_id;
  end if;
  update public.demandes
     set statut = 'refuse', solde_apres = v_solde, decided_at = current_date,
         decided_by = auth.uid(), motif_refus = coalesce(p_motif, '')
   where id = p_id;
end $$;

create or replace function public.remettre_en_attente(p_id uuid) returns void
language plpgsql security definer set search_path = public as $$
declare d public.demandes;
begin
  if not public.is_gestionnaire() then raise exception 'Accès refusé.' using errcode = '42501'; end if;
  select * into d from public.demandes where id = p_id for update;
  if not found then raise exception 'Demande introuvable (supprimée entre-temps ?).'; end if;
  if d.statut = 'attente' then return; end if;
  if d.statut = 'approuve' and d.type = 'annuel' then
    update public.agents set solde = solde + d.duree where id = d.agent_id;
  end if;
  -- Une demande refusée qui redevient active peut entrer en conflit (contrainte de chevauchement).
  update public.demandes
     set statut = 'attente', solde_apres = null, decided_at = null, decided_by = null, motif_refus = ''
   where id = p_id;
end $$;

create or replace function public.supprimer_demande(p_id uuid) returns void
language plpgsql security definer set search_path = public as $$
declare d public.demandes;
begin
  if not public.is_gestionnaire() then raise exception 'Accès refusé.' using errcode = '42501'; end if;
  select * into d from public.demandes where id = p_id for update;
  if not found then return; end if;
  if d.statut = 'approuve' and d.type = 'annuel' then
    update public.agents set solde = solde + d.duree where id = d.agent_id;
  end if;
  delete from public.demandes where id = p_id;
end $$;

revoke execute on function public.approuver_demande(uuid), public.refuser_demande(uuid, text),
  public.remettre_en_attente(uuid), public.supprimer_demande(uuid), public.is_gestionnaire()
  from public, anon;
grant execute on function public.approuver_demande(uuid), public.refuser_demande(uuid, text),
  public.remettre_en_attente(uuid), public.supprimer_demande(uuid), public.is_gestionnaire()
  to authenticated;

-- ---------------------------------------------------------------------
-- Temps réel : chaque gestionnaire voit les changements des autres immédiatement
-- ---------------------------------------------------------------------
do $$ begin
  alter publication supabase_realtime add table public.agents;
exception when duplicate_object then null; end $$;
do $$ begin
  alter publication supabase_realtime add table public.demandes;
exception when duplicate_object then null; end $$;
